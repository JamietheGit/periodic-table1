@st.cache_data
def load_element_data():
    """Load element data from JSON file."""
    data_path = "elements_data.json"
    if os.path.exists(data_path):
        with open(data_path, "r", encoding="utf-8") as f:
            return json.load(f)
    return {}

def get_element_by_atomic_number(atomic_num):

def get_element_by_symbol(symbol):
    for key, val in ELEMENTS_DATA.items():

def draw_bohr_diagram(element):
    """Dynamically generate a Matplotlib Bohr-style atomic diagram."""
    fig, ax = plt.subplots(figsize=(6, 6))
    fig.patch.set_facecolor('#0b1329')
    ax.set_facecolor('#0b1329')

    shells = element.get("shell_distribution", [2])
    symbol = element["symbol"]
    protons = element["protons"]
    
    # Standard shell maximum capacities
    max_shell_capacities = [2, 8, 18, 32, 32, 18, 8]

    # Draw nucleus
    nucleus_circle = plt.Circle((0, 0), 1.2, color='#1e3a8a', ec='#60a5fa', lw=2, zorder=10)
    ax.add_patch(nucleus_circle)
    ax.text(0, 0, f"{symbol}\n({protons}p)", color='white', weight='bold', fontsize=12, ha='center', va='center', zorder=11)

    max_shells = len(shells)
    base_radius = 2.2
    step = 1.4

    for i, occupied in enumerate(shells):
        r = base_radius + i * step
        max_cap = max_shell_capacities[i] if i < len(max_shell_capacities) else (2 * (i+1)**2)

        # Draw shell circle (unoccupied path model representation)
        shell_circle = plt.Circle((0, 0), r, color='#334155', fill=False, ls='--', lw=1.5, alpha=0.6)
        ax.add_patch(shell_circle)

        # Distribute electrons around the shell
        for e_idx in range(max_cap):
            angle = (2 * np.pi * e_idx) / max_cap - np.pi / 2
            ex = r * np.cos(angle)
            ey = r * np.sin(angle)

            if e_idx < occupied:
                # Occupied electron: solid blue dot
                dot = plt.Circle((ex, ey), 0.22, color='#38bdf8', ec='#bae6fd', lw=1.2, zorder=5)
                ax.add_patch(dot)
            else:
                # Unoccupied electron position: hollow circle with contrasting outline
                hollow = plt.Circle((ex, ey), 0.18, color='#0b1329', ec='#64748b', lw=1.5, zorder=4)
                ax.add_patch(hollow)

    limit = base_radius + max_shells * step + 0.8
    ax.set_xlim(-limit, limit)
    ax.set_ylim(-limit, limit)
    ax.axis('off')

    # Legend / Title
    ax.set_title(f"Bohr Model for {element['name']} ({element['electron_config']})", color='#f8fafc', fontsize=12, pad=10, weight='bold')
    
    return fig

# Helper function to render an element card in the grid
def render_element_tile(atomic_num):
    el = get_element_by_atomic_number(atomic_num)
    if not el:
        st.empty()
        return

    is_selected = (int(atomic_num) == st.session_state.selected_atomic_num)
    color = FAMILY_COLORS.get(el["family"], "#3b82f6")
    
    border_style = "border: 3px solid #f8fafc;" if is_selected else f"border: 1px solid {color};"
    bg_style = f"background-color: {color}33;" if is_selected else "background-color: #111e38;"

    button_label = f"**{el['symbol']}**\n{el['atomic_number']}"
    
    if st.button(f"{el['atomic_number']}: {el['symbol']} ({el['name']})", key=f"periodic_{atomic_num}", use_container_width=True):
        st.session_state.selected_atomic_num = atomic_num
        st.session_state.selected_family = el["family"]
        st.rerun()
